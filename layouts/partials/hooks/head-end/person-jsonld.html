{{/* Schema.org "Person" structured data for the homepage, built from the superuser's author profile. */}} {{- if .IsHome -}} {{- with index (first 1 (where (where site.Pages "Section" "authors") "Params.superuser" true)) 0 -}} {{- $person := dict "@context" "https://schema.org" "@type" "Person" "name" .Title "url" site.BaseURL -}} {{- with .Params.first_name }}{{ $person = merge $person (dict "givenName" .) }}{{ end -}} {{- with .Params.last_name }}{{ $person = merge $person (dict "familyName" .) }}{{ end -}} {{- with .Params.role }}{{ $person = merge $person (dict "jobTitle" .) }}{{ end -}} {{- with .Params.bio }}{{ $person = merge $person (dict "description" .) }}{{ end -}} {{- with (.Resources.ByType "image").GetMatch "avatar*" }}{{ $person = merge $person (dict "image" (.Fill "270x270 Center").Permalink) }}{{ end -}} {{- with index (.Params.organizations | default slice) 0 }}{{ $person = merge $person (dict "worksFor" (dict "@type" "Organization" "name" .name "url" .url)) }}{{ end -}} {{- $alumni := slice -}} {{- range .Params.education.courses }}{{ $alumni = $alumni | append (dict "@type" "CollegeOrUniversity" "name" .institution) }}{{ end -}} {{- with $alumni }}{{ $person = merge $person (dict "alumniOf" .) }}{{ end -}} {{- with .Params.interests }}{{ $person = merge $person (dict "knowsAbout" .) }}{{ end -}} {{- $same := slice -}} {{- range .Params.social }}{{ if hasPrefix .link "http" }}{{ $same = $same | append .link }}{{ end }}{{ end -}} {{- with $same }}{{ $person = merge $person (dict "sameAs" .) }}{{ end }} {{- end -}} {{- end -}}
